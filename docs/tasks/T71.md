### 1. `src/lib/cases.ts` 新增并导出 COLL / SBLS 规则谓词与规则构造

在 `src/lib/cases.ts` 中复用现有几何坐标（`S = STICKERS`），导出以下供 `COLL` / `SBLS` 直接挂载到 `SET_RULES` 与 `lintCase` 的纯函数与规则对象：

```ts
/** COLL 显示与完成判定掩码：F2L 33 张 + 顶层四角 12 张 + 顶面中心与顶层四棱顶面 5 张（共 50 张，仅排除顶层四棱侧面 FU/RU/BU/LU 4 张） */
export const isCollMask: StickerMask = (id) => isF2LPiece(id) || isCorner(id) || uFace(id);

/** 顶层四棱的 U 面贴纸位置（UB/UR/UF/UL 的 U 面，共 4 个位置） */
export const COLL_U_EDGE_TOP_LOCS: readonly number[] = S.filter(
  (s) => s.face === 'U' && !isCorner(s.id) && (s.p[0] !== 0 || s.p[2] !== 0),
).map((s) => s.id);

/** 顶层四角的 U 面贴纸位置（UBL/UBR/UFR/UFL 的 U 面，共 4 个位置） */
export const COLL_U_CORNER_TOP_LOCS: readonly number[] = S.filter(
  (s) => s.face === 'U' && isCorner(s.id),
).map((s) => s.id);

/** SBLS 必须保持不变的贴纸：左桥 1x2x3 全部 13 张 + 右桥后半（DR/DBR/BR/R中心）8 张 = 21 张（即 isRouxBlock 扣除 FR 槽 5 张） */
export const isSblsPreserve: StickerMask = (id) => isRouxBlock(id) && !isFRPair(id);

/** SBLS 目标槽位（DFR 角块 3 张 + FR 棱块 2 张 = 5 个贴纸位置） */
export const SBLS_FR_PAIR_LOCS: readonly number[] = S.filter((s) => isFRPair(s.id)).map((s) => s.id);

export const COLL_RULE: SetRule = {
  view: 'top',
  display: isCollMask,
  solve: isCollMask,
  preserve: isF2LPiece,
  preUs: [0, 1, 2, 3],
  preY: [0, 1, 2, 3],
  normalize: true,
};

export const SBLS_RULE: SetRule = {
  view: 'iso',
  display: isRouxBlock,
  solve: isRouxBlock,
  preserve: isSblsPreserve,
  preUs: [0],
  preY: [0],
  normalize: false,
};
```

同时导出供 `lintCase` 调用的附加不变量检查函数（在 `COLL` 和 `SBLS` 接入 `SetId` 时分别在 `lintCase` 中调用，本任务先对合成状态做单测）：

```ts
/**
 * 校验 COLL 附加不变量：
 * 1. 起手状态顶层四棱必须朝向正确（EO 已完成：COLL_U_EDGE_TOP_LOCS 上的贴纸 face 均为 'U'）；
 * 2. 起手状态顶层四角不能已经全部朝上（否则属于 PLL / EPLL，不属于 40 个 COLL）；
 * 3. 主公式 algs[0] 不得包含整体转体 x/y/z。
 */
export function lintCollState(caseId: string, after: CubeState, primaryAlg: string): string[] {
  const errors: string[] = [];
  if (!COLL_U_EDGE_TOP_LOCS.every((loc) => S[after[loc]].face === 'U')) {
    errors.push(`${caseId}: COLL prerequisite violated — U-layer edges are not oriented`);
  }
  if (COLL_U_CORNER_TOP_LOCS.every((loc) => S[after[loc]].face === 'U')) {
    errors.push(`${caseId}: COLL state has all 4 U-corners already oriented (PLL/O case)`);
  }
  if (parseAlg(primaryAlg).some((m) => 'xyz'.includes(m.base))) {
    errors.push(`${caseId}: COLL algs[0] must not contain whole-cube rotations`);
  }
  return errors;
}

/**
 * 校验 SBLS 附加不变量：
 * 1. 起手状态 FR 槽（DFR 角块 + FR 棱块）不能已经处于还原状态；
 * 2. 所有公式不得包含整体转体 x/y/z。
 */
export function lintSblsState(caseId: string, after: CubeState, algs: readonly string[]): string[] {
  const errors: string[] = [];
  if (SBLS_FR_PAIR_LOCS.every((loc) => after[loc] === loc)) {
    errors.push(`${caseId}: SBLS FR pair is already solved`);
  }
  for (const alg of algs) {
    if (parseAlg(alg).some((m) => 'xyz'.includes(m.base))) {
      errors.push(`${caseId}: SBLS alg (${alg}) must not contain whole-cube rotations`);
    }
  }
  return errors;
}
```
